\begin{abstract}
We derive the TAO-pure lending cycle
\textbf{open short $\to$ pump $\to$ close $\to$ unstake} from the
implemented pallet math (lift fraction, Balancer weights, invariant,
force-sell, readd). Fees are an explicit additive parameter, set to
zero in the base case. The lift fraction is denoted $\varphi$ and is
\textbf{not} a fee. The buggy close path (readd via weight
re-derivation at the post-pump mark) predicts the clone SN29 drains
(+289\,TAO / +122\,TAO). Weight-preserving readd flips the sign.
Under keep-weights we state an explicit bounded-search pass/fail
criterion (no $\le N$-action self-contained sequence yields attacker
net TAO $>0$ for $\varphi$ up to the side-cap) and report results on
live reserves. Lifted liquidity deliberately shallows the book:
slippage scales as $\approx 1/(1-\varphi)$ while a loan is open.
$\mu$'s September~2026 note is cited only as a cross-check after our
closed forms are in hand.
\end{abstract}

\section{Balancer and lift primitives}
\label{sec:primitives}

\subsection{Swaps (stake / unstake)}
At fixed $(w_b,w_q)$, fee $\varepsilon$ on input:
\begin{align}
\label{eq:sell}
\text{sell }\Delta x:\quad
\Delta y &= y\left(1-\Big(\frac{x}{x+\Delta x(1-\varepsilon)}\Big)^{w_b/w_q}\right),\\
\label{eq:buy}
\text{buy with }\Delta y:\quad
\Delta x &= x\left(1-\Big(\frac{y}{y+\Delta y(1-\varepsilon)}\Big)^{w_q/w_b}\right).
\end{align}
Weights are \textbf{not} updated mid-swap.

\subsection{Lift}
\texttt{lift\_protocol\_liquidity}$(\varphi)$ removes
$(\varphi x,\varphi y)$ with weights unchanged
(\texttt{pallets/swap/.../lift\_protocol\_liquidity}):
\[
(x,y,L)\mapsto\bigl((1-\varphi)x,(1-\varphi)y,(1-\varphi)L\bigr),\qquad
p\text{ unchanged}.
\]
For a short, pallet sets $\varphi=P/x$ so the alpha leg equals principal.

\subsection{Force-sell (short open)}
Sell $P=\varphi x$ into the post-lift book. At $\varepsilon=0$,
$x$ returns to its pre-lift value and
\begin{equation}
\label{eq:ys}
y_s
=
y\,(1-\varphi)^{1/w_q},
\qquad
E
=
(1-\varphi)y-y_s
=
y\Bigl[(1-\varphi)-(1-\varphi)^{1/w_q}\Bigr].
\end{equation}
(Proof: post-lift reserves $((1-\varphi)x,(1-\varphi)y)$; sell
$\varphi x$ gives
$y_s=(1-\varphi)y\cdot(1-\varphi)^{w_b/w_q}
=y(1-\varphi)^{1/w_q}$ since $1+w_b/w_q=1/w_q$.)
The sell preserves $L$ at $\varepsilon=0$, so
$L_s=(1-\varphi)L$.

\subsection{EMA band and LTV (feasibility only)}
Open rejects if post-trade spot exits
$\mathrm{EMA}\cdot[1-\beta,1+\beta]$ on the trade's side
(\texttt{ensure\_open\_ema\_band}). Short LTV requires
$C\ge\lambda\cdot(\text{debt mark})$. Side capacity caps
$\varphi y$ (lift TAO). These bound reachable $\varphi$; they do not
appear inside the pool algebra of a feasible open.

\subsection{Readd}
Custody returns $(\varphi y,P)=(\varphi y,\varphi x)$.
\begin{itemize}
  \item \textbf{Buggy} (pre-fix):
    \texttt{adjust\_protocol\_liquidity}$\to$
    \texttt{update\_weights\_for\_added\_liquidity} at the
    \emph{current} reserves $(x_c,y_c)$:
    \begin{equation}
    \label{eq:reweight}
    w_q'
    =
    \frac{w_q\,x_c\,(y_c+\varphi y)}
         {w_b\,y_c\,(x_c+\varphi x)+w_q\,x_c\,(y_c+\varphi y)},
    \qquad
    w_b'=1-w_q'.
    \end{equation}
    Spot is held at the post-pump mark; $L$ generally jumps.
  \item \textbf{Fixed} (design (a)): credit $(\varphi y,\varphi x)$
    with $(w_b,w_q)$ untouched.
\end{itemize}

\section{Cycle: short $\to$ pump $\to$ wallet repay $\to$ unstake}
\label{sec:cycle}

\begin{cycle}[Two-wallet drain shape (clone a13)]
\label{cyc:a13}
Attacker posts collateral $C$, opens a short of lift fraction $\varphi$,
receives escrow $E$. A pumper buys alpha with $S$ TAO. Attacker buys
$B$ TAO of alpha (enough to cover $P$), closes with
\texttt{repay\_from\_wallet}, then both unstake residual alpha.
\end{cycle}

\subsection{Base case $\varepsilon=0$}
\begin{enumerate}
  \item \textbf{Open.} Lift $\varphi$, force-sell $P=\varphi x$:
        state $(x,y_s)$ with $y_s$ as in \eqref{eq:ys}; escrow $E$;
        $L_s=(1-\varphi)L$; weights unchanged.
  \item \textbf{Pump $S$.} Buy with $S$ via \eqref{eq:buy} at
        $\varepsilon=0$:
        \[
          y_p=y_s+S,\qquad
          x_p=x\left(\frac{y_s}{y_p}\right)^{w_q/w_b},\qquad
          \alpha_{\mathrm{pump}}=x-x_p.
        \]
        $L$ unchanged ($L_p=L_s$).
  \item \textbf{Buy $B$ for repay.}
        $y_b=y_p+B$,
        $x_b=x_p\cdot(y_p/y_b)^{w_q/w_b}$,
        $\alpha_{\mathrm{buy}}=x_p-x_b$ (need $\alpha_{\mathrm{buy}}\ge P$).
  \item \textbf{Wallet repay + readd.}
        Burn $P$ alpha from wallet; refund $E+C$; credit
        $(\varphi y,P)$ into $(x_b,y_b)$.
        \begin{itemize}
          \item Buggy: apply \eqref{eq:reweight}; get $(w_b',w_q')$,
                reserves $(x_b+\varphi x,\,y_b+\varphi y)$, new
                $L'= (x_b+\varphi x)^{w_b'}(y_b+\varphi y)^{w_q'}$.
          \item Keep: same reserve credit, weights stay $(w_b,w_q)$.
        \end{itemize}
  \item \textbf{Unstake} residual alpha
        $\alpha_{\mathrm{pump}}+(\alpha_{\mathrm{buy}}-P)$ via
        \eqref{eq:sell} at the post-readd weights.
\end{enumerate}

\subsection{Attacker PnL}
Combined free of attacker + pumper (collateral is internal):
\begin{equation}
\label{eq:pnl}
\Delta
=
-S-B+E+\underbrace{\Delta y_{\mathrm{unstake}}}_{\text{sell residual }\alpha}.
\end{equation}
When the residual sell restores $x$ near baseline, empirically
$\Delta\approx y_0-y_{\mathrm{end}}=-\Delta\texttt{SubnetTAO}$.
Profit iff $\Delta>0$.

\subsection{Production fee}
Replace every swap with \eqref{eq:sell}--\eqref{eq:buy} at
$\varepsilon=33/65535$. Lift / readd / weight update unchanged.
No fee is charged on lift or on the weight update itself.

\begin{theorem}[Drain $=$ weight re-derivation on readd]
\label{thm:drain}
On the observed SN29 states, the open$\to$pump$\to$close$\to$unstake
cycle has $\Delta>0$ if and only if readd uses
\eqref{eq:reweight} at the post-pump mark. With weights held fixed,
$\Delta<0$ (fee-lossy). Fees are a small correction; the base-case
($\varepsilon=0$) buggy path is already profitable.
\end{theorem}

\begin{remark}[Theorem status]
Theorem~\ref{thm:drain} is a closed-form / numeric identity on the
observed SN29 clone books. A general closed-form
$\Delta\le 0$ for \emph{every} keep-weights cycle is still open.
Section~\ref{sec:criterion} supplies the operational pass/fail
surrogate used for shipping.
\end{remark}

\section{Root fix}
\label{sec:fix}

\texttt{readd\_protocol\_liquidity} credits $(\Delta y,\Delta x)$
\textbf{without} calling \eqref{eq:reweight}. Emissions still use
\texttt{adjust\_protocol\_liquidity}. Stake/unstake unchanged.

This is not ``Uniswap~v2'': $w_q\neq\tfrac12$ stays a weighted
Balancer CPMM; only lending readd skips asymmetric-join reweight
(Section~\ref{sec:univ2}).

\section{Pass/fail criterion (keep-weights)}
\label{sec:criterion}

\begin{definition}[Atomic action]
An \emph{atomic action} is one of:
\texttt{open\_short}, \texttt{open\_long},
\texttt{buy}/\texttt{AddStake},
\texttt{sell}/\texttt{RemoveStake},
\texttt{close} (wallet repay or escrow buyback).
Force-sell / force-buy inside open, and protocol readd on close, are
not separate attacker actions. A $k$-way equal split of one buy or
sell counts as $k$ actions.
\end{definition}

\begin{definition}[Self-contained sequence]
A sequence is \emph{self-contained} if it starts and ends with no open
loan on the subnet and with all attacker-controlled alpha inventory
unwound to TAO. Multi-coldkey means the attacker may use several
coldkeys (e.g.\ a dedicated pumper); attacker net TAO $\Delta$ is the
sum of free-balance changes across those keys.
\end{definition}

\begin{definition}[Side-cap $\varphi_{\max}$]
With no open book, $G=y$. Default \texttt{LiftParams} give
\[
  \mathrm{Cap}
  =
  \max\bigl(f\cdot G,\;\min(G-y^\star,\,0.5\,G)\bigr),
  \qquad
  \mathrm{Cap}_{\mathrm{short}}=0.5\cdot\mathrm{Cap},
\]
where $f=0.33$ if $G\ge 3000\,\tau$ else $f=0.15$, and
$y^\star=\mathrm{ref}/((1.10)^{w_b}-1)$.
Short $\varphi_{\max}=\mathrm{Cap}_{\mathrm{short}}/y$;
long $\varphi_{\max}=\mathrm{Cap}_{\mathrm{long}}/y$.
\end{definition}

\begin{definition}[Bounded-search pass/fail]
\label{def:pass}
Fix $N\in\mathbb{N}$ (default $N=8$) and production fee
$\varepsilon=33/65535$. On each live reserve snapshot, under
\textbf{keep-weights} readd:
\[
  \textbf{PASS}
  \iff
  \nexists\text{ self-contained sequence of length }\le N
  \text{ with }\varphi\in(0,\varphi_{\max}]
  \text{ and attacker }\Delta > 0.
\]
The search \textbf{must} include multi-coldkey and trade-split
variants. The same grid run against buggy readd must find
$\Delta>0$ on at least one cell (sanity that the harness detects the
known drain).
\end{definition}

\paragraph{Search status.}
Script \texttt{scripts/no\_arbitrage\_cycle\_search.py} on live
finney snapshots (SN29, SN64~Chutes, SN51, SN4, SN119 thin) with
$N=8$, up to 3-way splits, multi-coldkey on/off, short and long,
wallet and buyback closes, $\varphi$ up to each side-cap
($\varphi_{\max}\in\{0.165,0.25,0.25,0.25,0.075\}$):
keep-weights reports \textbf{PASS} on $20\,087$ live cells
(max $\Delta\approx -0.01\,\tau$); buggy is profitable on
$16\,619$ cells of the same grid (sanity). JSON:
\texttt{no\_arbitrage\_cycle\_search.json}. Closed-form coverage
beyond Theorem~\ref{thm:drain} remains open.

\section{Deliberate volatility: slippage $\approx 1/(1-\varphi)$}
\label{sec:slippage}

A short lift removes $(\varphi x,\varphi y)$ into custody at unchanged
spot, leaving executable reserves $((1-\varphi)x,(1-\varphi)y)$.
(The subsequent force-sell then moves the mark to
$y_s=y(1-\varphi)^{1/w_q}$ at $\varepsilon=0$; that is a price move,
not the depth claim.) Relative to the un-lifted book, a small trade of
fixed TAO size therefore sees fractional price impact scaled by
\begin{equation}
\label{eq:slippage}
M(\varphi)\;\approx\;\frac{1}{1-\varphi}.
\end{equation}
This is intentional: lending lifts liquidity out of the AMM for the
life of the loan and makes the pool shallower. Figure~\ref{fig:slippage}
plots $M(\varphi)$ up to each subnet's $\varphi_{\max}$ (side-cap),
with the cap marked.

\begin{figure}[ht]
\centering
\includegraphics[width=0.92\textwidth]{no-arbitrage-slippage.pdf}
\caption{Slippage multiplier $M(\varphi)=1/(1-\varphi)$ vs lift
fraction for live SN29, SN64~(Chutes), SN51, SN4 and thin SN119, each
truncated at its short side-cap $\varphi_{\max}=\mathrm{Cap}_{\mathrm{short}}/y$.
Script: \texttt{no\_arbitrage\_slippage\_chart.py}.}

\begin{remark}
We do \textbf{not} import $\mu$'s closed form as an assumption.
Equation~\eqref{eq:ys} is proved from lift + force-sell in
Section~\ref{sec:primitives}. That it coincides with $\mu$'s
$(1-\varphi)^{1/w_2}$ term is the cross-check. Our profitability
statement is $\Delta>0$ in \eqref{eq:pnl} under buggy readd --- an
attacker convention --- whereas $\mu$'s $F<1$ is a pool-cost
convention on a fee-free short cycle without the weight-ratchet term.
The SN29 drain is the ratchet \eqref{eq:reweight}, which that
September fee-free $F$ does not encode.
\end{remark}

\section{Fixed weights are not Uniswap v2}
\label{sec:univ2}

Main re-derives weights on \emph{asymmetric liquidity adds}
(emissions), so those adds do not move spot. Swaps never re-derived
weights. Uni~v2 is the special case $w_b=w_q=\tfrac12$. Holding
$w_q\approx 0.47$ fixed is still weighted Balancer. Design~(a) loses
only price-neutrality of \emph{lending readd after skew}; that is the
intentional root fix.

\paragraph{Recommendation (Jacob).}
Ship~(a). Optional hardening: restore $\Delta L$ by a current-ratio
proportional join and mark-to-market the custody residual (design~(b)).
Do not restore current-mark \eqref{eq:reweight} on lending readd.
